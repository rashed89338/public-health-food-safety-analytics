/*------------------------------------------------------------
  12. Weighted descriptive checks
------------------------------------------------------------*/

proc surveyfreq data=cleandata.restaurant_appended;
    cluster psu_no;
    weight pw;
    tables time * inside_pump_drink / row col;
run;

proc surveyfreq data=cleandata.restaurant_appended;
    cluster psu_no;
    weight pw;
    tables time * improved_toi / row col;
run;
